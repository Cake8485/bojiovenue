/* ==========================================================================
   Charts — hand-rolled SVG so there is no chart library to ship or keep updated.
   A page registers a spec in CHART_SPECS and drops <div class="chart" data-chart="id">;
   drawCharts() (called after every render and on resize) paints them at real width.
   ========================================================================== */
const CHART_SPECS = {};
const short = (n) => { const a = Math.abs(n); return (n < 0 ? '-' : '') + (a >= 1000 ? (a / 1000).toFixed(a % 1000 === 0 ? 0 : 1) + 'k' : Math.round(a)); };
const niceMax = (v) => { if (v <= 0) return 1; const p = Math.pow(10, Math.floor(Math.log10(v))); const f = v / p; return (f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 5 ? 5 : 10) * p; };

function drawCharts() {
  $$('[data-chart]').forEach((el) => {
    const s = CHART_SPECS[el.dataset.chart]; if (!s) return;
    el.innerHTML = (s.type === 'line' ? lineChart(el, s) : s.type === 'bars' ? barChart(el, s) : donutChart(el, s)).s;
  });
}

function axes(W, H, pad, min, max, fmt) {
  const ticks = 4; let out = '<g class="grid">';
  for (let i = 0; i <= ticks; i++) {
    const v = min + ((max - min) * i) / ticks, y = pad.t + (1 - i / ticks) * (H - pad.t - pad.b);
    out += `<line x1="${pad.l}" x2="${W - pad.r}" y1="${y}" y2="${y}"/><text x="${pad.l - 8}" y="${y + 3}" text-anchor="end">${esc((fmt || short)(v))}</text>`;
  }
  return out + '</g>';
}
function xLabels(labels, xAt, W, pad, H, minGap, sub) {
  const plot = W - pad.l - pad.r, k = Math.max(1, Math.ceil(labels.length / Math.max(1, Math.floor(plot / (minGap || 34)))));
  return labels.map((l, i) => {
    if (!(i % k === 0 || i === labels.length - 1)) return '';
    return `<text x="${xAt(i)}" y="${sub ? H - 17 : H - 6}" text-anchor="middle">${esc(l)}</text>` + (sub && sub[i] ? `<text x="${xAt(i)}" y="${H - 5}" text-anchor="middle" style="fill:var(--ink4)">${esc(sub[i])}</text>` : '');
  }).join('');
}

function lineChart(el, s) {
  const W = Math.max(240, el.clientWidth || 560), H = s.h || 210, pad = { l: 46, r: 16, t: 14, b: s.sub ? 40 : 28 };
  const n = s.labels.length, all = s.series.flatMap((x) => x.values);
  const hi = niceMax(Math.max(...all, 0)), lo = Math.min(0, ...all) < 0 ? -niceMax(-Math.min(...all)) : 0;
  const xAt = (i) => pad.l + (n === 1 ? (W - pad.l - pad.r) / 2 : (i * (W - pad.l - pad.r)) / (n - 1));
  const yAt = (v) => pad.t + (1 - (v - lo) / (hi - lo || 1)) * (H - pad.t - pad.b);
  let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="${esc(s.aria || 'Chart')}"><defs>`;
  s.series.forEach((se, k) => { svg += `<linearGradient id="g${el.dataset.chart}${k}" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="${se.color}" stop-opacity=".2"/><stop offset="1" stop-color="${se.color}" stop-opacity="0"/></linearGradient>`; });
  svg += '</defs>' + axes(W, H, pad, lo, hi, s.fmt);
  s.series.forEach((se, k) => {
    const pts = se.values.map((v, i) => `${xAt(i).toFixed(1)},${yAt(v).toFixed(1)}`);
    if (se.area && n > 1) svg += `<path d="M${pts.join('L')}L${xAt(n - 1)},${yAt(0)}L${xAt(0)},${yAt(0)}Z" fill="url(#g${el.dataset.chart}${k})"/>`;
    svg += n > 1 ? `<path d="M${pts.join('L')}" fill="none" stroke="${se.color}" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"${se.dash ? ' stroke-dasharray="4 4"' : ''}/>` : '';
    const last = se.values.length - 1;
    if (se.dot !== false) svg += `<circle cx="${xAt(last)}" cy="${yAt(se.values[last])}" r="4.5" fill="${se.color}" stroke="var(--card)" stroke-width="2"/>`;
  });
  (s.incomplete || []).forEach((flag, i) => { if (flag && i < n - 1 && s.series[0]) svg += `<circle cx="${xAt(i)}" cy="${yAt(s.series[0].values[i])}" r="4" fill="var(--card)" stroke="${s.series[0].color}" stroke-width="2"/>`; });
  svg += xLabels(s.labels, xAt, W, pad, H, 0, s.sub);
  const bw = n === 1 ? W - pad.l - pad.r : (W - pad.l - pad.r) / (n - 1);
  for (let i = 0; i < n; i++) {
    const tip = `${(s.tips || s.labels)[i]}\n` + s.series.map((se) => `${se.name}: ${(s.fmtTip || money)(se.values[i])}`).join('\n') + (s.extraTip ? '\n' + (s.extraTip(i) || '') : '');
    svg += `<rect class="hit" x="${xAt(i) - bw / 2}" y="${pad.t}" width="${bw}" height="${H - pad.t - pad.b}" data-tip="${esc(tip)}"/>`;
  }
  return raw(svg + '</svg>');
}

function barChart(el, s) {
  const W = Math.max(240, el.clientWidth || 560), H = s.h || 220, pad = { l: 46, r: 12, t: 12, b: s.sub ? 40 : 28 };
  const n = s.labels.length, groups = s.series.length, plotW = W - pad.l - pad.r, slot = plotW / n;
  const totalsAt = s.labels.map((_, i) => (s.stacked ? sumBy(s.series, (x) => x.values[i]) : Math.max(...s.series.map((x) => x.values[i]), 0)));
  const hi = niceMax(Math.max(...totalsAt, 0)), lo = 0;
  const yAt = (v) => pad.t + (1 - (v - lo) / (hi - lo || 1)) * (H - pad.t - pad.b);
  let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="${esc(s.aria || 'Chart')}">` + axes(W, H, pad, lo, hi, s.fmt);
  const bw = Math.min(34, (slot * 0.62) / (s.stacked ? 1 : groups));
  s.labels.forEach((l, i) => {
    const cx = pad.l + slot * i + slot / 2; let acc = 0;
    s.series.forEach((se, k) => {
      const v = se.values[i] || 0; if (v <= 0) return;
      const x = s.stacked ? cx - bw / 2 : cx - (bw * groups) / 2 + k * bw, y0 = s.stacked ? yAt(acc + v) : yAt(v), hgt = (s.stacked ? yAt(acc) : yAt(0)) - y0;
      svg += `<rect x="${x.toFixed(1)}" y="${y0.toFixed(1)}" width="${(bw - (s.stacked ? 0 : 2)).toFixed(1)}" height="${Math.max(1, hgt).toFixed(1)}" rx="3" fill="${se.color}"/>`;
      if (s.stacked) acc += v;
    });
    if (s.incomplete && s.incomplete[i]) svg += `<text x="${cx}" y="${yAt(totalsAt[i]) - 6}" text-anchor="middle" style="fill:var(--amber);font:700 12px var(--mono)">?</text>`;
    const tip = `${(s.tips || s.labels)[i]}\n` + s.series.map((se) => `${se.name}: ${money(se.values[i] || 0)}`).join('\n') + (s.stacked ? `\nTotal: ${money(totalsAt[i])}` : '') + (s.extraTip ? '\n' + (s.extraTip(i) || '') : '');
    svg += `<rect class="hit" x="${pad.l + slot * i}" y="${pad.t}" width="${slot}" height="${H - pad.t - pad.b}" data-tip="${esc(tip)}" ${s.click ? `data-act="${s.click}" data-i="${i}" style="cursor:pointer"` : ''}/>`;
  });
  svg += xLabels(s.labels, (i) => pad.l + slot * i + slot / 2, W, pad, H, 0, s.sub);
  return raw(svg + '</svg>');
}

function donutChart(el, s) {
  const size = s.size || 176, R = size / 2 - 14, C = 2 * Math.PI * R, total = sumBy(s.parts, (p) => p.value) || 1;
  let off = 0, svg = `<svg viewBox="0 0 ${size} ${size}" role="img" aria-label="${esc(s.aria || 'Breakdown')}" style="max-width:${size}px;margin:0 auto">`;
  svg += `<circle cx="${size / 2}" cy="${size / 2}" r="${R}" fill="none" stroke="var(--grey-bg)" stroke-width="20"/>`;
  s.parts.filter((p) => p.value > 0).forEach((p) => {
    const len = (p.value / total) * C;
    svg += `<circle cx="${size / 2}" cy="${size / 2}" r="${R}" fill="none" stroke="${p.color}" stroke-width="20" stroke-dasharray="${Math.max(0, len - 1.5).toFixed(2)} ${(C - len + 1.5).toFixed(2)}" stroke-dashoffset="${(-off).toFixed(2)}" transform="rotate(-90 ${size / 2} ${size / 2})" data-tip="${esc(`${p.name}: ${money(p.value)} (${Math.round((p.value / total) * 100)}%)`)}"/>`;
    off += len;
  });
  if (s.center) svg += `<text x="${size / 2}" y="${size / 2 + 2}" text-anchor="middle" style="font:700 20px var(--sans);fill:var(--ink);text-transform:none;letter-spacing:-.01em">${esc(s.center.big)}</text><text x="${size / 2}" y="${size / 2 + 18}" text-anchor="middle">${esc(s.center.small || '')}</text>`;
  return raw(svg + '</svg>');
}
