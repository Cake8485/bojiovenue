const ACCENT = 'var(--accent)', WARM = 'var(--warm)', POS = 'var(--pos)', MUTE = 'var(--ink4)', GOLD = '#C99700';
const REGIONS = {};   // re-renderable table regions (search boxes live OUTSIDE them so typing never loses focus)

const head = (title, sub, actions) => h`<div class="page-head"><div><h1>${title}</h1><div class="sub label">${sub}</div></div><div class="head-actions">${actions || ''}</div></div>`;
const cta = (label, hover, icon, act, data) => h`<button class="cta" data-act="${act}" ${raw(data || '')}>${ico(icon, icon === 'send' ? 'fly' : '')}<span class="lab"><span class="l1">${label}</span><span class="l2">${hover}</span></span></button>`;
const kpi = (label, icon, value, ctx, tone, alert) => h`<div class="card kpi ${alert ? 'alert' : ''}"><div class="top-row"><span class="label">${label}</span>${ico(icon)}</div><div class="n">${value}</div><div class="ctx ${tone || ''}">${ctx}</div></div>`;
const emptyState = (title, text) => h`<div class="empty"><img src="/dashboard/brand/mascot.png" alt=""><b>${title}</b><span>${text}</span></div>`;
const accordion = (id, title, meta, body, open) => h`<details class="card acc" id="${id}" ${open ? raw('open') : ''}><summary>${ico('chev')}<span>${title}</span><span class="meta">${meta}</span></summary><div class="acc-b">${body}</div></details>`;
const pill = (text, tone) => h`<span class="pill ${tone || ''}">${text}</span>`;
const stateTone = { signed: 'pos', awaiting: 'amber', confirmed: 'brand', recorded: '', tentative: 'dash' };
const stateText = { signed: 'Signed', awaiting: 'Awaiting signature', confirmed: 'Confirmed', recorded: 'Recorded', tentative: 'Soft book' };
const payPill = (b) => (!b.counted ? pill('Awaiting signature', 'amber') : b.paid_state === 'paid' ? pill('Paid', 'pos') : b.paid_state === 'partial' ? pill('Partly paid', 'amber') : pill('Unpaid', 'neg'));
const depTone = { refunded: 'pos', partial: 'amber', forfeited: 'neg', held: 'brand' };
const depText = { refunded: 'Refunded', partial: 'Partly refunded', forfeited: 'Forfeited', held: 'Held' };
// chart axis for a run of months: 3-letter month, the year underneath Jan / the first month, the full name in the tooltip,
// and a flag for months with confirmed bookings but no recorded amounts (drawn as ? / hollow points)
const monthSpec = (ms) => ({ labels: ms.map((m) => fmtMon(m.month)), sub: ms.map((m, i) => (i === 0 || m.month.endsWith('-01') ? m.month.slice(0, 4) : '')), tips: ms.map((m) => fmtMonth(m.month)), incomplete: ms.map((m) => m.unpriced > 0) });
const gapNote = (ms) => (ms.some((m) => m.unpriced > 0) ? raw('<p class="muted" style="font-size:12px;margin-top:8px"><b style="color:var(--amber)">?</b> / hollow point = confirmed bookings that month have no recorded amounts, so it is understated — not a quiet month.</p>') : '');
const waLink = (phone, text) => { const dg = String(phone || '').replace(/\D/g, ''); return `https://wa.me/${dg.length === 8 ? '65' + dg : dg}?text=${encodeURIComponent(text)}`; };
const firstName = (n) => String(n || '').trim().split(/\s+/).slice(0, 2).join(' ');

function banner() {
  const warns = S.data.health.filter((x) => x.level === 'warn'); if (!warns.length) return '';
  return h`<div class="banner">${ico('alert')}<span><b>Some figures are incomplete.</b> ${warns.slice(0, 2).map((w) => w.title).join(' · ')}${warns.length > 2 ? ` · +${warns.length - 2} more` : ''}</span><button data-act="open-health">See what’s missing</button></div>`;
}

function table(id, cols, rows, opts = {}) {
  const st = S.sort[id]; let data = rows.slice();
  if (st) { const c = cols.find((x) => x.k === st.k); if (c && c.sort) data.sort((a, b) => { const x = c.sort(a), y = c.sort(b); return (x < y ? -1 : x > y ? 1 : 0) * st.dir; }); }
  if (!data.length) return opts.empty || emptyState('Nothing here yet', '');
  return h`<div class="table-wrap"><table class="t ${opts.cls || ''}"><thead><tr>${cols.map((c) => h`<th class="${c.r ? 'r' : ''}" ${c.sort ? raw(`data-sort="${esc(c.k)}" data-act="sort" data-t="${esc(id)}"`) : ''}>${c.label}${st && st.k === c.k ? (st.dir > 0 ? ' ↑' : ' ↓') : ''}</th>`)}</tr></thead>
    <tbody>${data.map((row) => h`<tr ${opts.rowAct ? raw(`data-act="${opts.rowAct}" data-key="${esc(row.key || row.id || row.name)}"`) : ''}>${cols.map((c) => h`<td class="${c.r ? 'r' : ''}">${c.f(row)}</td>`)}</tr>`)}</tbody>
    ${opts.foot ? h`<tfoot>${opts.foot}</tfoot>` : ''}</table></div>`;
}
function rerender(region) { const el = $('#reg-' + region); if (el && REGIONS[region]) { el.innerHTML = REGIONS[region]().s; drawCharts(); } }

/* ---------- Overview ---------- */
function actionItems() {
  const d = S.data, t = d.today, out = [];
  const od = overdueRefunds(); if (od) out.push({ tone: 'neg', t: `Refund ${plural(od.count, 'deposit')} — overdue`, m: 'REFUNDS', act: 'go', arg: 'refunds' });
  const owed = d.bookings.filter((b) => b.counted && b.outstanding > 0 && b.source === 'app');
  if (owed.length) out.push({ tone: 'warn', t: `Collect ${money(sumBy(owed, (b) => b.outstanding))} from ${plural(owed.length, 'client')}`, m: 'TO COLLECT', act: 'collect' });
  d.calendar.filter((e) => e.state === 'awaiting' && e.date >= t).slice(0, 2).forEach((e) => out.push({ tone: 'warn', t: `Chase signature — ${e.client}`, m: `IN ${daysBetween(t, e.date)}D`, act: 'cal-date', arg: e.date }));
  d.calendar.filter((e) => e.date >= t && daysBetween(t, e.date) <= 7 && (e.state === 'signed' || e.state === 'confirmed')).slice(0, 3)
    .forEach((e) => out.push({ tone: '', t: `${e.client} — ${fmtShort(e.date)}${e.time ? ' · ' + e.time : ''}`, m: daysBetween(t, e.date) === 0 ? 'TODAY' : `IN ${daysBetween(t, e.date)}D`, act: 'cal-date', arg: e.date }));
  const coll = [...new Set(d.calendar.filter((e) => e.collision && e.date >= t).map((e) => e.date))];
  coll.slice(0, 2).forEach((dt) => out.push({ tone: 'neg', t: `Two bookings on ${fmtShort(dt)} — check for a double-booking`, m: 'CLASH', act: 'cal-date', arg: dt }));
  const miss = d.health.find((x) => x.id === 'missing_amounts');
  if (miss) out.push({ tone: 'warn', t: `Record amounts for ${plural(miss.count, 'past booking')}`, m: 'MISSING DATA', act: 'add-booking' });
  return out.slice(0, 7);
}

function healthList() {
  const hs = S.data.health;
  if (!hs.length) return h`<div class="empty"><b>Everything is accounted for</b></div>`;
  return h`${hs.map((x) => h`<div class="health-item"><span class="dot" style="margin-top:7px;background:${x.level === 'warn' ? 'var(--warm)' : 'var(--ink4)'}"></span><div><b>${x.title}</b><p>${x.detail}</p>${x.action ? h`<button class="btn sm" style="margin-top:8px" data-act="${x.action.do}">${x.action.label}</button>` : ''}</div></div>`)}
    <p class="muted" style="font-size:12.5px;margin-top:10px">Rule: gaps are stated here, never filled in with estimates. Source of truth stays accounting.xlsx, expenses-2026.xlsx and the Lead Manager.</p>`;
}

  CHART_SPECS['ex-cat'] = { type: 'donut', aria: 'Costs by category', parts: cats, center: { big: money(t.cost), small: 'COSTS' } };
