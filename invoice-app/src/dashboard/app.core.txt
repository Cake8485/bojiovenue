async function api(path, opts = {}) {
  const r = await fetch('/api/dashboard/' + path, { ...opts, headers: { Authorization: 'Bearer ' + S.key, 'Content-Type': 'application/json', ...(opts.headers || {}) } });
  if (r.status === 401) { signOut('Wrong key, or your session expired. Enter it again.'); throw new Error('unauthorized'); }
  const text = await r.text();
  let j = {}; try { j = JSON.parse(text); } catch (e) { /* csv etc. */ }
  if (!r.ok) throw new Error(j.error || `Something went wrong (${r.status}).`);
  return opts.raw ? text : j;
}
async function downloadCsv(kind, label) {
  try {
    const text = await api('export/' + kind, { raw: true });

    document.body.appendChild(a); a.click(); a.remove();

}

const monthAdd = (m, k) => { let y = +m.slice(0, 4), mo = +m.slice(5, 7) - 1 + k; y += Math.floor(mo / 12); mo = ((mo % 12) + 12) % 12; return `${y}-${String(mo + 1).padStart(2, '0')}`; };

function range() {
  const t = S.data.today, cm = t.slice(0, 7), first = S.data.months.length ? S.data.months[0].month : cm;
  const label = (PERIODS.find((p) => p[0] === S.period) || [0, ''])[1];
  switch (S.period) {
    case 'this': return { from: cm + '-01', to: t, label };
    case 'last': return { from: monthAdd(cm, -1) + '-01', to: lastDay(monthAdd(cm, -1)), label };
    case '3m': return { from: monthAdd(cm, -2) + '-01', to: t, label };
    case 'ytd': return { from: t.slice(0, 4) + '-01-01', to: t, label };
    case 'all': return { from: first + '-01', to: t, label };
    case 'custom': return { from: S.custom.from || first + '-01', to: S.custom.to || t, label: `${fmtDate(S.custom.from || first + '-01')} – ${fmtDate(S.custom.to || t)}` };
    default: return { from: monthAdd(cm, -11) + '-01', to: t, label: 'Last 12 months' };
  }
}

  function paint() {
    const q = input.value.trim();
    shown = all.map((it) => ({ it, s: fuzzy(q, it.label) })).filter((x) => x.s > 0 && (q || !it0(x.it))).sort((a, b) => b.s - a.s).slice(0, 14).map((x) => x.it);
    if (!q) shown = all.filter((x) => !x.hint).slice(0, 12);
    sel = Math.min(sel, Math.max(0, shown.length - 1));
    let g = null, out = '';
    shown.forEach((it, i) => { if (it.g !== g) { out += `<div class="sec label">${esc(it.g)}</div>`; g = it.g; } out += h`<button class="row" role="option" data-i="${i}" aria-selected="${i === sel}">${ico(it.icon)}<span>${it.label}</span>${it.k ? h`<kbd>${it.k}</kbd>` : ''}</button>`.s; });
    res.innerHTML = out || '<div class="empty">Nothing matches.</div>';
  }
  const it0 = (it) => !!it.hint;
  const run = (i) => { const it = shown[i]; if (!it) return; closeOverlay(); lastFocus = null; setTimeout(it.run, 0); };
  input.addEventListener('input', () => { sel = 0; paint(); });
  input.addEventListener('keydown', (e) => {
    if (e.key === 'ArrowDown') { sel = Math.min(shown.length - 1, sel + 1); paint(); e.preventDefault(); }
    else if (e.key === 'ArrowUp') { sel = Math.max(0, sel - 1); paint(); e.preventDefault(); }
    else if (e.key === 'Enter') { run(sel); e.preventDefault(); }
  });
  res.addEventListener('click', (e) => { const b = e.target.closest('.row'); if (b) run(+b.dataset.i); });
  res.addEventListener('mousemove', (e) => { const b = e.target.closest('.row'); if (b && +b.dataset.i !== sel) { sel = +b.dataset.i; $$('.row', res).forEach((r, i) => r.setAttribute('aria-selected', i === sel)); } });
  paint(); input.focus();
}

/* ---------- keyboard ---------- */
document.addEventListener('keydown', (e) => {
  const typing = /^(INPUT|TEXTAREA|SELECT)$/.test((e.target || {}).tagName || '');
  if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') { e.preventDefault(); if ($('#pq')) closeOverlay(); else openPalette(); return; }
  if (e.key === 'Escape' && $('#overlay-root').firstChild) { closeOverlay(); return; }
  if (typing || e.metaKey || e.ctrlKey || e.altKey || !S.data || $('#overlay-root').firstChild) return;
  const k = e.key.toLowerCase();
  if (k === '/') { e.preventDefault(); openPalette(); }
  else if (k === 'e') ACT['add-expense']();
  else if (k === 'b') ACT['add-booking']();
  else if (k === 'c' && S.data.summary.to_collect > 0) ACT.collect();
});
