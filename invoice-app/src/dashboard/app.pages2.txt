const goto = (id) => { if (location.hash === '#/' + id) route(); else location.hash = '#/' + id; };

/* ---------- Clients ---------- */
REGIONS.clients = () => {
  const q = S.f.clients.q.trim().toLowerCase();
  const rows = S.data.clients.filter((c) => !q || [c.name, ...c.aliases].join(' ').toLowerCase().includes(q));
  return table('cli', [
    { k: 'name', label: 'Client', sort: (c) => c.name.toLowerCase(), f: (c) => h`<b>${c.name}</b>${c.repeat ? raw(' <span class="pill brand">repeat</span>') : ''}<span class="sub">${c.kind}</span>` },
    { k: 'n', label: 'Bookings', r: 1, sort: (c) => c.bookings, f: (c) => c.bookings },
    { k: 'rev', label: 'Revenue', r: 1, sort: (c) => c.revenue, f: (c) => (c.priced_count ? money(c.revenue) + (c.priced_count < c.bookings ? '+' : '') : h`<span class="muted" data-tip="No amounts recorded for this client">—</span>`) },
    { k: 'last', label: 'Last event', sort: (c) => c.last || '', f: (c) => (c.last ? fmtDate(c.last) : '—') },
    { k: 'next', label: 'Next event', sort: (c) => c.next || '9999', f: (c) => (c.next ? h`<b>${fmtDate(c.next)}</b>` : '—') },
  ], rows, { rowAct: 'client-open', empty: emptyState('No clients match', '') });
};

/* ---------- Calendar ---------- */
const isoDow = (iso) => new Date(iso + 'T00:00:00Z').getUTCDay();
const addDay = (iso, k) => new Date(Date.parse(iso + 'T00:00:00Z') + k * 86400000).toISOString().slice(0, 10);
const isPremium = (iso) => { const w = isoDow(iso), H = S.data.holidays; return w === 5 || w === 6 || w === 0 || H.includes(iso) || H.includes(addDay(iso, 1)); };
PAGES.calendar = () => {
  const d = S.data, { y, m } = S.cal, mm = String(m).padStart(2, '0'), ym = `${y}-${mm}`;
  const dow = (new Date(Date.UTC(y, m - 1, 1)).getUTCDay() + 6) % 7, dim = new Date(Date.UTC(y, m, 0)).getUTCDate(), rows = Math.ceil((dow + dim) / 7);
  const start = addDay(`${ym}-01`, -dow), byDate = {};
  d.calendar.forEach((e) => { (byDate[e.date] = byDate[e.date] || []).push(e); });
  let cells = '';
  for (let i = 0; i < rows * 7; i++) {
    const iso = addDay(start, i), inMonth = iso.startsWith(ym), ev = byDate[iso] || [], ph = d.holidays.includes(iso);
    cells += h`<div class="cal-day ${inMonth ? '' : 'off'} ${iso === d.today ? 'today' : ''} ${inMonth && isPremium(iso) ? 'prem' : ''}"><div class="dn"><span>${iso === d.today ? raw(`<b>${+iso.slice(8)}</b>`) : +iso.slice(8)}</span>${ph ? raw('<span class="ph">PH</span>') : ''}</div>
      ${ev.slice(0, 3).map((e) => h`<button class="cal-chip ${e.state} ${e.collision ? 'collision' : ''}" data-act="cal-chip" data-key="${e.key}" data-tip="${e.client}${e.time ? ' · ' + e.time : ''}${e.collision ? '\nCheck: two bookings on this date' : ''}">${e.time ? e.time.slice(0, 5) + ' ' : ''}${e.client}</button>`)}
      ${ev.length > 3 ? h`<button class="cal-chip confirmed" data-act="cal-day" data-date="${iso}">+${ev.length - 3} more</button>` : ''}</div>`.s;
  }
  const upcoming = d.calendar.filter((e) => e.date >= d.today && e.state !== 'tentative').slice(0, 8);
  const inMonthEvents = d.calendar.filter((e) => e.date.startsWith(ym) && e.state !== 'tentative');
  return h`${head('Confirmed Booking Calendar', `${plural(d.summary.upcoming_events, 'upcoming booking')} · premium days shaded · PH = public holiday`, cta('Record booking', 'Add a booking', 'plus', 'add-booking'))}
    <div class="card"><div class="cal-head"><h2>${MONTHS[m - 1]} ${y}</h2><span class="label">${plural(inMonthEvents.length, 'booking')}</span>
      <button class="btn btn-icon sm" data-act="cal-prev" aria-label="Previous month"><span style="display:inline-block;transform:rotate(180deg)">${ico('chev')}</span></button>
      <button class="btn sm" data-act="cal-today">Today</button><button class="btn btn-icon sm" data-act="cal-next" aria-label="Next month">${ico('chev')}</button></div>
      <div class="cal-grid">${['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'].map((x) => h`<div class="cal-dow">${x}</div>`)}${raw(cells)}</div>
      <div class="cal-legend"><span class="pill pos">Signed</span><span class="pill amber">Awaiting signature</span><span class="pill brand">Confirmed (Lead Manager / accounts)</span><span class="pill dash">Soft book</span><span class="pill neg">Red outline = two bookings, one date</span></div></div>
    <div class="card" style="margin-top:14px"><div class="card-h"><h2>Coming up</h2></div><div class="card-b">${upcoming.length ? table('cal-up', [
      { k: 'd', label: 'Date', f: (e) => fmtDate(e.date) }, { k: 'c', label: 'Client', f: (e) => h`${e.client}${e.collision ? raw(' <span class="pill neg">clash</span>') : ''}` }, { k: 't', label: 'Time', f: (e) => e.time || '—' },
      { k: 'k', label: 'Type', f: (e) => e.kind }, { k: 's', label: 'Status', f: (e) => pill(stateText[e.state], stateTone[e.state]) }], upcoming, { rowAct: 'cal-date-row' }) : emptyState('Nothing booked ahead', 'Confirmed bookings appear here.')}</div></div>`;
};
const eventByKey = (key) => S.data.calendar.find((e) => e.key === key);
function openEvent(e) {
  if (!e) return;
  const b = S.data.bookings.find((x) => x.key === e.key);
  openDrawer({ title: e.client, body: h`<dl class="kv"><dt>Date</dt><dd>${fmtDate(e.date)}${S.data.holidays.includes(e.date) ? raw(' <span class="pill amber">Public holiday</span>') : ''}</dd><dt>Time</dt><dd>${e.time || '—'}${e.hours ? ` (${e.hours}h)` : ''}</dd>
    <dt>Type</dt><dd>${e.kind}${e.purpose ? ' · ' + e.purpose : ''}</dd><dt>Status</dt><dd>${pill(stateText[e.state], stateTone[e.state])}</dd>
    <dt>Source</dt><dd>${e.source === 'app' ? 'Invoice app' : e.source === 'ledger' ? 'Accounts workbook / entered by hand' : 'Lead Manager'}</dd>
    <dt>Revenue</dt><dd>${e.revenue == null ? raw('<span class="muted">No amounts recorded</span>') : money(e.revenue)}</dd></dl>
    ${e.collision ? h`<div class="banner" style="margin-top:16px">${ico('alert')}<span>Another booking is confirmed on this date — check it isn’t a double-booking.</span></div>` : ''}`,
    footer: h`${b ? h`<button class="btn" data-act="booking-open" data-key="${b.key}">Open booking</button>` : h`<button class="btn" data-act="add-booking" data-date="${e.date}" data-client="${e.client}" data-purpose="${e.purpose || ''}">${ico('plus')}Record amounts</button>`}
      <button class="btn" data-act="client-open" data-key="${(S.data.clients.find((c) => c.name === e.client || c.aliases.includes(e.client)) || {}).name || ''}">Client</button>` });
}
ACT['cal-chip'] = (el) => openEvent(eventByKey(el.dataset.key));
ACT['cal-date-row'] = (el) => openEvent(eventByKey(el.dataset.key));
ACT['cal-day'] = (el) => { const evs = S.data.calendar.filter((e) => e.date === el.dataset.date); openDrawer({ title: fmtDate(el.dataset.date), body: h`<div class="list">${evs.map((e) => h`<button class="item" data-act="cal-chip" data-key="${e.key}"><span class="ring"></span><span class="t">${e.client}</span><span class="m">${e.time || ''}</span></button>`)}</div>` }); };
ACT['cal-date'] = (el) => { const date = el.dataset.arg; S.cal = { y: +date.slice(0, 4), m: +date.slice(5, 7) }; goto('calendar'); setTimeout(() => { const evs = S.data.calendar.filter((e) => e.date === date); if (evs.length === 1) openEvent(evs[0]); else if (evs.length) ACT['cal-day']({ dataset: { date } }); }, 30); };
ACT['cal-prev'] = () => { S.cal = S.cal.m === 1 ? { y: S.cal.y - 1, m: 12 } : { y: S.cal.y, m: S.cal.m - 1 }; renderPage(); };
ACT['cal-next'] = () => { S.cal = S.cal.m === 12 ? { y: S.cal.y + 1, m: 1 } : { y: S.cal.y, m: S.cal.m + 1 }; renderPage(); };
ACT['cal-today'] = () => { const t = S.data.today; S.cal = { y: +t.slice(0, 4), m: +t.slice(5, 7) }; renderPage(); };

async function loadTemplates() {
  const box = $('#tpl-box'); if (!box) return;
  try {
    const { templates } = await api('templates'); const map = Object.fromEntries(templates.map((t) => [t.key, t.body]));
    box.innerHTML = Object.keys(TPL).map((k) => h`<label class="field"><span>${TPL[k][0]} — ${TPL[k][1]}</span><textarea id="tpl-${k}" maxlength="1000">${map[k] || ''}</textarea><div><button class="btn sm" data-act="save-template" data-key="${k}">Save</button></div></label>`.s).join('');
  } catch (e) { box.innerHTML = h`<p class="err">${e.message}</p>`.s; }
}
ACT['save-template'] = async (el) => { const k = el.dataset.key; try { await api('templates/' + k, { method: 'PUT', body: JSON.stringify({ body: $('#tpl-' + k).value }) }); toast('Message saved'); } catch (e) { toast(e.message, true); } };
ACT['set-theme'] = (el) => { S.theme = el.dataset.v; applyTheme(); renderPage(); };
ACT.signout = () => signOut('Signed out.');

      <dt>Type</dt><dd>${b.kind}${b.purpose ? ' · ' + b.purpose : ''}</dd><dt>Status</dt><dd>${payPill(b)}</dd></dl>

      <dt><b>Revenue</b></dt><dd><b>${money(b.revenue)}</b></dd><dt>Collected</dt><dd>${money(b.collected)}</dd><dt>Balance</dt><dd>${b.outstanding > 0 ? h`<b style="color:var(--neg)">${money(b.outstanding)}</b>` : '—'}</dd>

    footer: h`${b.source === 'ledger' && b.editable ? h`<button class="btn danger" data-act="del-booking" data-id="${b.row_id}">Delete</button>` : ''}

};
ACT.collect = () => {
  const rows = S.data.bookings.filter((b) => b.counted && b.outstanding > 0 && b.source === 'app').sort((a, b) => (a.date < b.date ? -1 : 1));
  openDrawer({ title: `Collect ${money(sumBy(rows, (b) => b.outstanding))}`, body: rows.length ? h`<div class="list">${rows.map((b) => {
      const msg = `Hi ${firstName(b.client)}, a friendly reminder that ${money(b.outstanding)} is still outstanding for your BoJioVenue booking on ${fmtDate(b.date)}. Thank you! 😊`;
      return h`<div class="item" style="cursor:default"><span class="ring neg"></span><span><span class="t">${b.client}</span><span class="muted" style="font-size:12.5px">${fmtDate(b.date)} · ${b.booking_no}${b.date < S.data.today ? ' · event has passed' : ''}</span></span>
        <span class="m"><b style="color:var(--neg)">${money(b.outstanding)}</b></span>${b.phone ? h`<a class="btn sm" data-native="1" target="_blank" rel="noopener" href="${waLink(b.phone, msg)}" aria-label="Remind ${b.client} on WhatsApp">${ico('wa')}</a>` : h`<span class="pill dash">no phone</span>`}</div>`; })}</div>
    <p class="muted" style="font-size:12.5px;margin-top:14px">The WhatsApp button opens a pre-written reminder — you tap send. Nothing goes out automatically.</p>` : emptyState('Nothing to collect', 'Every invoice-app booking is paid up.') });
};
ACT['del-booking'] = async (el) => { if (!confirm('Delete this booking record? This can’t be undone.')) return; try { await api('ledger-bookings/' + el.dataset.id, { method: 'DELETE' }); closeOverlay(); await load(true); toast('Booking record deleted'); } catch (e) { toast(e.message, true); } };

/* ---------- expenses: form ---------- */
function openExpense(v, id) {
  const cats = S.data.categories;
  openDrawer({ title: id ? 'Edit expense' : 'Add expense', body: h`<form id="f-exp" autocomplete="off">
    <div class="two">${fld('Date', 'date', `type="date" required value="${esc(v.date || S.data.today)}"`)}${fld('Amount ($)', 'amount', `type="number" step="0.01" min="0.01" required value="${esc(v.amount || '')}"`)}</div>
    <label class="field"><span>Category</span><select name="category">${cats.map((c) => h`<option ${c === (v.category || 'Utilities') ? raw('selected') : ''}>${c}</option>`)}</select></label>
    ${fld('Description', 'description', `required maxlength="200" placeholder="e.g. October utilities bill" value="${esc(v.description || '')}"`)}
    <div class="two"><label class="field"><span>Paid by</span><input name="paid_by" list="payers" value="${esc(v.paid_by || '')}"><datalist id="payers"><option>Kenneth</option><option>Carine</option></datalist></label>
      <label class="field"><span>Payment method</span><select name="payment_method"><option value="">—</option>${['PayNow', 'Bank transfer', 'Cash', 'Card', 'Other'].map((m) => h`<option ${m === v.payment_method ? raw('selected') : ''}>${m}</option>`)}</select></label></div>
    ${fld('Receipt file', 'receipt_file', `placeholder="2026-11-18_vendor_what-it-is.jpg" value="${esc(v.receipt_file || '')}"`)}
    <label class="field"><span>Notes</span><textarea name="notes" maxlength="500">${v.notes || ''}</textarea></label>
    <label class="field row"><input type="checkbox" name="is_estimate" ${v.estimate ? raw('checked') : ''}><span style="text-transform:none;letter-spacing:0;font:inherit;color:var(--ink2)">This is an estimate — the real bill isn’t on file yet</span></label><div class="err" id="f-err"></div></form>`,
    footer: h`${id ? h`<button class="btn danger" type="button" data-act="del-expense" data-id="${id}">Delete</button>` : ''}<button class="btn" type="button" data-act="close">Cancel</button><button class="btn primary" type="submit" form="f-exp">Save</button>`,
    onMount: () => { $('#f-exp').addEventListener('submit', async (e) => { e.preventDefault(); const fd = Object.fromEntries(new FormData(e.target)); fd.is_estimate = !!e.target.is_estimate.checked;
      try { await api(id ? 'expenses/' + id : 'expenses', { method: id ? 'PUT' : 'POST', body: JSON.stringify(fd) }); closeOverlay(); await load(true); toast(id ? 'Expense updated' : 'Expense added'); } catch (err) { $('#f-err').textContent = err.message; } }); } });
}
ACT['add-expense'] = (el) => { const ds = (el && el.dataset) || {}; openExpense({ category: ds.cat, amount: ds.amt, description: ds.desc, estimate: ds.est === '1' }); };
ACT['edit-expense'] = (el) => { const e = S.data.expenses.find((x) => x.id === +el.dataset.id); if (e) openExpense({ date: e.date, amount: e.amount, category: e.category, description: e.description, paid_by: e.paid_by, payment_method: e.payment_method, receipt_file: e.receipt_file, notes: e.notes, estimate: e.is_estimate }, e.id); };
ACT['del-expense'] = async (el) => { if (!confirm('Delete this expense? This can’t be undone.')) return; try { await api('expenses/' + el.dataset.id, { method: 'DELETE' }); closeOverlay(); await load(true); toast('Expense deleted'); } catch (e) { toast(e.message, true); } };
